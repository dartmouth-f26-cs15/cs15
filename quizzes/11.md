1. What does the following program do? What's the bug in it?

```c
#include <stddef.h>   // for NULL
#include <sys/mman.h> // for mmap, MAP_*, PROT_*
#include <stdlib.h>   // for EXIT_FAILURE

int main(void) {
    char *const p = mmap(
        NULL,                        // addr
        1,                           // len
        PROT_READ,                   // prot
        MAP_ANONYMOUS | MAP_PRIVATE, // flags
        -1,                          // fd
        0,                           // offset
    )
    if (p == MAP_FAILED) {
        return EXIT_FAILURE;
    }
    p[0] = 'a';
    return p[0];
}
```
